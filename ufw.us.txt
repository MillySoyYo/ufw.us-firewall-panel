sudo bash -c 'cat << "EOF" > /usr/local/bin/ufw.us
#!/usr/bin/env bash

C_RESET="\033[0m"
C_YELLOW="\033[1;33m"
C_RED="\033[1;31m"
C_CYAN="\033[1;35m"

check_permissions() {
    if [ "$EUID" -ne 0 ]; then
        echo -e "${C_RED}Error: Administrator privileges are required to run this panel.${C_RESET}"
        echo -e "${C_RED}Please run with sudo or as root (e.g., sudo ufw.us).${C_RESET}"
        exit 1
    fi
}

check_system() {
    check_permissions
    if ! command -v ufw &> /dev/null; then
        echo -e "${C_RED}Error: Official UFW is not installed on this system.${C_RESET}"
        exit 1
    fi
}

execute_ufw_command() {
    local cmd="$1"
    echo -e "Executing command: ufw $cmd"
    
    local output
    local exit_code
    
    output=$(ufw $cmd 2>&1)
    exit_code=$?
    
    echo ""
    if [ $exit_code -eq 0 ]; then
        echo -e "${C_CYAN}Command executed successfully.${C_RESET}"
        echo -e "$output"
    else
        echo -e "${C_RED}ERROR: UFW command failed with exit code $exit_code.${C_RESET}"
        echo -e "$output"
    fi
}

press_back() {
    echo -e ""
    read -p "Press Enter to return..."
}

view_rules() {
    clear
    echo -e "${C_CYAN}=== Live UFW Firewall Rules ===${C_RESET}"
    echo ""
    
    local output
    local exit_code
    output=$(ufw status verbose 2>&1)
    exit_code=$?
    
    if [ $exit_code -eq 0 ]; then
        echo -e "$output"
    else
        echo -e "${C_RED}ERROR: Failed to retrieve UFW status (exit code $exit_code).${C_RESET}"
        echo -e "$output"
    fi
    press_back
}

validate_port_input() {
    local input="$1"
    if [[ "$input" =~ ^([0-9]{1,5})$ ]]; then
        if [ "$input" -ge 1 ] && [ "$input" -le 65535 ]; then
            return 0
        fi
    elif [[ "$input" =~ ^([0-9]{1,5}):([0-9]{1,5})$ ]]; then
        local start="${BASH_REMATCH[1]}"
        local end="${BASH_REMATCH[2]}"
        if [ "$start" -ge 1 ] && [ "$start" -le 65535 ] && [ "$end" -ge 1 ] && [ "$end" -le 65535 ] && [ "$start" -le "$end" ]; then
            return 0
        fi
    fi
    return 1
}

add_port_rule() {
    while true; do
        clear
        echo -e "${C_CYAN}=== Add Port Rule ===${C_RESET}"
        echo ""
        echo "0. Back"
        echo ""
        echo -e "${C_YELLOW}WARNING:${C_RESET}"
        echo -e "${C_YELLOW}A large port range may open many ports.${C_RESET}"
        echo -e "${C_YELLOW}Only use a range when you fully understand the security impact.${C_RESET}"
        echo ""
        read -p "Enter Port or Port Range (e.g., 443 or 443:65535): " port_input
        if [ "$port_input" = "0" ]; then
            return
        fi

        if ! validate_port_input "$port_input"; then
            echo -e "${C_RED}Invalid port or range format/range (Valid: 1-65535).${C_RESET}"
            sleep 1.5
            continue
        fi

        echo ""
        echo "Select Protocol:"
        echo "1. TCP"
        echo "2. UDP"
        echo "0. Back"
        read -p "Choose option [0-2]: " proto_choice

        if [ "$proto_choice" = "0" ]; then
            return
        fi

        local proto=""
        case "$proto_choice" in
            1) proto="tcp" ;;
            2) proto="udp" ;;
            *)
                echo -e "${C_RED}Invalid protocol selection.${C_RESET}"
                sleep 1.5
                continue
                ;;
        esac

        echo ""
        echo -e "${C_YELLOW}WARNING:${C_RESET}"
        echo -e "${C_YELLOW}Changing firewall rules may affect server access.${C_RESET}"
        echo ""
        echo -e "Port: ${C_CYAN}${port_input}${C_RESET}"
        echo -e "Protocol: ${C_CYAN}${proto^^}${C_RESET}"
        echo -e "Command action: ${C_CYAN}ufw allow ${port_input}/${proto}${C_RESET}"
        echo ""

        read -p "Proceed with adding rule? (y/N): " confirm
        if [[ "$confirm" =~ ^[Yy]$ ]]; then
            echo ""
            execute_ufw_command "allow ${port_input}/${proto}"
            press_back
            return
        else
            echo -e "${C_RED}Operation cancelled.${C_RESET}"
            press_back
            return
        fi
    done
}

delete_port_rule() {
    while true; do
        clear
        echo -e "${C_CYAN}=== Delete Port Rule ===${C_RESET}"
        echo ""
        echo "0. Back"
        echo ""
        echo -e "${C_YELLOW}WARNING:${C_RESET}"
        echo -e "${C_YELLOW}A large port range may open many ports.${C_RESET}"
        echo -e "${C_YELLOW}Only use a range when you fully understand the security impact.${C_RESET}"
        echo ""
        read -p "Enter Port or Port Range to delete (e.g., 443 or 443:65535): " port_input
        if [ "$port_input" = "0" ]; then
            return
        fi

        if ! validate_port_input "$port_input"; then
            echo -e "${C_RED}Invalid port or range format/range.${C_RESET}"
            sleep 1.5
            continue
        fi

        echo ""
        echo "Select Protocol:"
        echo "1. TCP"
        echo "2. UDP"
        echo "0. Back"
        read -p "Choose option [0-2]: " proto_choice

        if [ "$proto_choice" = "0" ]; then
            return
        fi

        local proto=""
        case "$proto_choice" in
            1) proto="tcp" ;;
            2) proto="udp" ;;
            *)
                echo -e "${C_RED}Invalid protocol selection.${C_RESET}"
                sleep 1.5
                continue
                ;;
        esac

        echo ""
        echo -e "${C_YELLOW}WARNING:${C_RESET}"
        echo -e "${C_YELLOW}Deleting firewall rules may affect running services or remote access.${C_RESET}"
        echo ""
        echo -e "Port: ${C_CYAN}${port_input}${C_RESET}"
        echo -e "Protocol: ${C_CYAN}${proto^^}${C_RESET}"
        echo -e "Command action: ${C_CYAN}ufw delete allow ${port_input}/${proto}${C_RESET}"
        echo ""

        read -p "Proceed with deletion? (y/N): " confirm
        if [[ "$confirm" =~ ^[Yy]$ ]]; then
            echo ""
            execute_ufw_command "delete allow ${port_input}/${proto}"
            press_back
            return
        else
            echo -e "${C_RED}Operation cancelled.${C_RESET}"
            press_back
            return
        fi
    done
}

main_menu() {
    check_system
    while true; do
        clear
        echo -e "${C_CYAN}========================================${C_RESET}"
        echo -e "${C_CYAN}    UFW Firewall Management Panel       ${C_RESET}"
        echo -e "${C_CYAN}========================================${C_RESET}"
        echo ""
        echo -e "${C_YELLOW}Type ufw.us to open this firewall panel${C_RESET}"
        echo ""
        echo "1. View Firewall Rules"
        echo "2. Add Port Rule"
        echo "3. Delete Port Rule"
        echo "4. Exit Panel"
        echo ""
        read -p "Enter your choice [1-4]: " choice

        case "$choice" in
            1) view_rules ;;
            2) add_port_rule ;;
            3) delete_port_rule ;;
            4) 
               clear
               echo "Exiting firewall panel. Goodbye!"
               exit 0 
               ;;
            *)
               echo -e "${C_RED}Invalid choice. Please select between 1 and 4.${C_RESET}"
               sleep 1
               ;;
        esac
    done
}

main_menu
EOF

chmod +x /usr/local/bin/ufw.us
'

echo -e "\033[1;32mDeployment complete! Run:\033[1;33m ufw.us\033[0m"
